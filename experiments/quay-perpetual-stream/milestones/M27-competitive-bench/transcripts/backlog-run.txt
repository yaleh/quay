$ backlog task edit task-1 --add-label bug --plain   # second, separate invocation

                          # within a single invocation, but DO accumulate across
                          # separate invocations

                          # -- narrows the finding to --add-label specifically on `edit`
                          # (repeated-flag parsing), not a general multi-label defect
